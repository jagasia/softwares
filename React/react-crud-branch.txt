import axios from "axios";

const URL="http://localhost:5001/branch";
class BranchService
{
    getAllBranches()
    {
        return axios.get(URL);
    }

    findBranchById(bid)
    {
        return axios.get(URL+"/"+bid);
    }

    addBranch(branch)
    {
        return axios.post(URL,branch);
    }

    updateBranch(branch)
    {
        return axios.put(URL,branch);
    }

    deleteBranch(bid)
    {
        return axios.delete(URL+"/"+bid);
    }
}

export default new BranchService;

import { Component } from "react";
import branchService from "../services/branch-service";

export default class BranchComponent extends Component
{
    constructor()
    {
        super();
        this.state={
            bid:'',
            bname:'',
            bcity:'',
            errors:{
                bid:'',
                bname:'',
                bcity:''
            }
        };
       

        this.handleChange=this.handleChange.bind(this);
        this.addBranch=this.addBranch.bind(this);
        this.updateBranch=this.updateBranch.bind(this);
        this.deleteBranch=this.deleteBranch.bind(this);
    }

    handleChange(event)
    {
        this.setState({[event.target.name]:event.target.value});
    }

    addBranch()
    {
        //check the state errors part. then only send the request to add
        branchService.addBranch(this.state);
    }
    
    updateBranch()
    {
        branchService.updateBranch(this.state);
    }

    deleteBranch()
    {
        branchService.deleteBranch(this.state.bid);
    }

    render()
    {
        return <div>
            <h2 className="text-center">Branch Form</h2>
            <form>
                Id: <input type="text" name="bid" className="form-control" onChange={this.handleChange} />
                Name: <input type="text" name="bname" className="form-control" onChange={this.handleChange} />
                City: <input type="text" name="bcity" className="form-control" onChange={this.handleChange} />
                <br/>
                <button type="button" className="btn btn-success" onClick={this.addBranch} >Add</button>&nbsp;&nbsp;&nbsp;
                <button type="button" className="btn btn-warning" onClick={this.updateBranch} >Update</button>&nbsp;&nbsp;&nbsp;
                <button type="button" className="btn btn-danger" onClick={this.deleteBranch} >Delete</button>
            </form>
            <br/>

        </div>
    }
}

import { Component } from "react";
import branchService from "../services/branch-service";

export default class DisplayBranch extends Component
{

    constructor()
    {
        super();
        this.state={branches:[]};
        branchService.getAllBranches().then((response)=>{
            this.setState({branches:response.data});
        }
            
        );
    }

    render()
    {
        return (<div>

<table className="table table-bordered table-striped table-hover">
                <thead>
                    <tr>
                        <th>Id</th>
                        <th>Name</th>
                        <th>City</th>
                    </tr>
                </thead>
                <tbody>
                    {
                        this.state.branches.map((branch)=>{
                            return(<tr>
                                <td>{branch.bid}</td>
                                <td>{branch.bname}</td>
                                <td>{branch.bcity}</td>
                            </tr>);
                        })
                    }
                </tbody>
            </table>

        </div>);
    }
}
